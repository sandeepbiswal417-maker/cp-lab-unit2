//p7.c Array of pointers to structures;-
//book search
#include<stdio.h>
struct Book
{
    int id;
    char title[20];
    char author[20];
};
    int main()
{
    struct Book b1={101,"C-Intro","Ritchie D"};
    struct Book b2={102,"story","AAA"};
    struct Book b3={103,"tab","XXX"};
    //Array of ptr to structures.
    struct Book *books[3]={&b1,&b2,&b3};
    int searchID,i,found=0;
    printf("enter Book 1D to search");
    scanf("%d",&searchID);
    for (i=0;i<3;i++)
    {
        if(books[i]->id==searchID)
        {found=1;
        printf("Book found");
        printf("1D=%d\n",books[i]->id);
        printf("title=%s\n",books[i]->title);
        printf("title=%s\n",books[i]->author);
        break;
        }
    }
    if(!found)
    printf("Not found");
    return 0;
}
